Integrasi Google Apps Script (Backend)

Salin kode di bawah ini ke editor Apps Script spreadsheet Anda untuk otomatisasi sync penuh!

Cara Pemasangan di Google Sheets:

  1. Buka Google Sheets KKN Anda di browser.
  2. Klik menu Ekstensi (Extensions) > Apps Script.
  3. Hapus semua isi file Code.gs lalu **Copy & Paste** kode JavaScript di bawah ini.
  4. Simpan () dan klik **Terapkan / Deploy** > **Penerapan Baru (New Deployment)** > Pilih jenis **Web App**.
  5. Akses diberikan ke **"Siapa Saja (Anyone)"**, lalu klik Deploy!

function doGet() {
  return HtmlService.createTemplateFromFile('Index')
      .evaluate()
      .setTitle('Sistem Keuangan KKN GRADASI 2026 KELOMPOK 9')
      .addMetaTag('viewport', 'width=device-width, initial-scale=1')

// Inisialisasi Sheet Keuangan Otomatis
function setupSheets() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  // 1. Sheet Transaksi
  var shTrans = ss.getSheetByName("TRANSAKSI") || ss.insertSheet("TRANSAKSI");
  if (shTrans.getLastRow() === 0) {
    shTrans.appendRow(["ID", "Tanggal", "Tipe", "Kategori", "Keterangan", "Inputor", "Nominal", "Link Bukti"]);
    shTrans.getRange("A1:H1").setFontWeight("bold").setBackground("#0284c7").setFontColor("#FFFFFF");
  }
  
  // 2. Sheet Kas Anggota
  var shKas = ss.getSheetByName("KAS_ANGGOTA") || ss.insertSheet("KAS_ANGGOTA");
  if (shKas.getLastRow() === 0) {
    shKas.appendRow(["ID Anggota", "Nama", "Divisi", "M1", "M2", "M3", "M4", "M5", "M6"]);
    shKas.getRange("A1:I1").setFontWeight("bold").setBackground("#0284c7").setFontColor("#FFFFFF");
  }

  return "Inisialisasi Sheet Berhasil!";
}

// Simpan Transaksi Baru dari Web UI
function addTransactionFromWeb(data) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("TRANSAKSI") || ss.insertSheet("TRANSAKSI");
  var newId = "TRX-" + new Date().getTime();
  
  sheet.appendRow([
    newId,
    data.tanggal,
    data.tipe,
    data.kategori,
    data.keterangan,
    data.inputor,
    data.nominal,
    data.bukti || "-"
  ]);
  
  return { status: "success", id: newId };
}

// Ambil Data Transaksi
function getTransactionsData() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("TRANSAKSI");
  if (!sheet) return [];
  
  var values = sheet.getDataRange().getValues();
  if (values.length <= 1) return [];
  
  var headers = values[0];
  var result = [];
  
  for (var i = 1; i < values.length; i++) {
    var row = values[i];
    result.push({
      id: row[0],
      tanggal: row[1],
      tipe: row[2],
      kategori: row[3],
      keterangan: row[4],
      inputor: row[5],
      nominal: row[6],
      bukti: row[7]
    });
  }
  return result;
}
